## Key Data Structure

```c
struct s2n_stuffer {
    struct s2n_blob blob;         /* underlying buffer */
    uint32_t read_cursor;
    uint32_t write_cursor;
    uint32_t high_water_mark;
    unsigned int alloced  : 1;    /* stuffer owns the blob memory */
    unsigned int growable : 1;    /* can the stuffer grow? */
    unsigned int tainted  : 1;    /* raw access occurred — prevents resize */
};
/* s2n_stuffer_data_available(s) == s->write_cursor - s->read_cursor */
/* s2n_stuffer_space_remaining(s) == s->blob.size - s->write_cursor */

struct s2n_blob {
    uint8_t *data;
    uint32_t size;
    uint32_t allocated;
    unsigned int growable : 1;
};
```
